#include <stdio.h>
#include "util.h"
#define BUFFER_SIZE 0x400 /* bytes */

typedef struct node {
    int value;
    struct node *next;
} node_t;

/** Sums the list. @param head first node */
static int sum(const node_t *head) {
    int total = 0;
    for (const node_t *n = head; n != NULL; n = n->next) {
        total += n->value;
    }
    return total;
}
