msg     db "Hello, world!", 10, 0
len     equ $ - msg
%define BUFSZ 0x100
%macro PRINT 1
    mov rdi, %1
%endmacro

section .text
global _start
_start:
    mov eax, 1          ; sys_write
    lea rsi, [rel msg]
    mov byte [rbx+4*rcx+8], 0FFh
    xor ecx, ecx
    vaddps ymm0, ymm1, [rdi]
    movaps xmm0, xmm1
    cmp eax, 0b101
    jne .done
.done:
    rep movsb
    syscall
    ret
    .globl main
    .type main, @function
main:
    pushq %rbp
    movl $0x10, %eax
    movq (%rax,%rbx,4), %rcx   # comment
    call printf@PLT
    jmp 1f
1:  ret

.code

    mov eax, OFFSET buf
    invoke ExitProcess, 0

    rjmp reset

    out DDRB, temp
    ldi r17, high(RAMEND)
    out SPH, r17

    lds r18, $0100
