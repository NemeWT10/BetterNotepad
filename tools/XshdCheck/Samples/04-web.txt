#[Attribute]
final class User extends Model implements JsonSerializable
{
    public const MAX = 10;
    private ?string $name = null;

    /** Greets. @param string $who */
    public function greet(string $who, int ...$rest): string
    {
        $msg = "Hello $who, {$this->name} \t ${who}";
        $raw = 'it\'s';
        $n = 0x1F + 1_000 + 1.5e3; // note
        if ($n >= 10 && !empty($who)) { return strtoupper($msg); }
        foreach ($rest as $k => $v) { echo $v, PHP_EOL; }
        return $this->name ?? self::MAX;
    }
}
