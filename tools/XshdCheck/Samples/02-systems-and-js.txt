/** A generic repo. */
@Injectable()
export class Repo<T extends { id: number }> implements IRepo<T> {
  private readonly items = new Map<number, T>();
  constructor(private readonly db: Database, public name?: string) {}

  async find(id: number): Promise<T | undefined> {
    const key: string = `item-${id}`;
    return this.items.get(id) ?? (await this.db.load<T>(key)) as T;
  }
}
